\section{SECURITY ANALYSIS}
\label{sec-security-analysis}
We next establish four mechanism-level guarantees corresponding to the security objectives in Section~\ref{subsec-security-objectives}. The analysis is intentionally scoped to the trust boundaries enforced by Algorithms~\ref{alg-patb}--\ref{alg-trust-policy}; it does not assume that the full nonlinear PCPS or the underlying MARL optimizer admits a global convergence proof.

\subsection{Bounded Poisoning Influence}
\begin{theorem}[Bounded poisoning influence]
\label{thm-bounded-poisoning}
Consider one assertion $q$ and two belief trajectories with the same initial log-odds, trusted prior, honest evidence, and independently verified feedback. Suppose the attacked trajectory additionally contains evidence from at most $f_p$ compromised provenance groups per update, each with group cap $\beta_g\le\beta_{max}$. Under $0<\lambda_f<1$, the adversarial log-odds deviation satisfies
\begin{align}
\label{eq-thm-logodds-bound}
|L_t^A(q)-L_t^0(q)|
\le \frac{f_p\beta_{max}}{1-\lambda_f},
\end{align}
and the corresponding belief deviation satisfies
\begin{align}
\label{eq-thm-belief-bound}
|b_t^A(q)-b_t^0(q)|
\le \frac{f_p\beta_{max}}{4(1-\lambda_f)}.
\end{align}
\end{theorem}

\begin{proof}
By Eq.~\ref{eq-patb-group-scale}, the total absolute contribution of one provenance group in one update is at most $\beta_g$. Hence the malicious contribution at time $t$ has magnitude at most $f_p\beta_{max}$. Let $\Delta_t=L_t^A(q)-L_t^0(q)$. Since the honest terms are identical in the two trajectories, Eq.~\ref{eq-patb-update} gives
\begin{align}
|\Delta_{t+1}|\le\lambda_f|\Delta_t|+f_p\beta_{max}.
\end{align}
With $\Delta_0=0$, unrolling the recursion yields
\begin{align}
|\Delta_t|\le f_p\beta_{max}\sum_{k=0}^{t-1}\lambda_f^k
\le\frac{f_p\beta_{max}}{1-\lambda_f}.
\end{align}
The sigmoid derivative is at most $1/4$, so $|\sigma(x)-\sigma(y)|\le|x-y|/4$, which gives Eq.~\ref{eq-thm-belief-bound}.
\end{proof}

The result shows that bounded evidence control produces bounded knowledge distortion even under repeated evidence injection; the attacker cannot increase influence merely by duplicating reports behind the same provenance group.

\subsection{False-Authorization Risk}
\begin{theorem}[False-authorization risk bound]
\label{thm-false-authorization}
Let action $\mathbf{a}$ require prerequisites $\mathcal{Q}(\mathbf{a})$. Assume each belief used by the semantic gate is calibrated up to $\varepsilon_q$, in the sense that whenever $b_t(q)\ge1-\delta_q$,
\begin{align}
\label{eq-calibration-assumption}
\Pr(q_t^\star=0\mid b_t(q)\ge1-\delta_q)
\le\delta_q+\varepsilon_q.
\end{align}
If Algorithm~\ref{alg-risk-gate} enforces Eqs.~\ref{eq-gate-prerequisite}--\ref{eq-gate-risk-budget} and the authoritative verifier correctly enforces invariant constraints, then
\begin{align}
\label{eq-thm-auth-bound}
\Pr(\mathbf{a}\ \text{is falsely authorized})
\le
\sum_{q\in\mathcal{Q}(\mathbf{a})}(\delta_q+\varepsilon_q)
\le\bar\delta_B+\sum_{q\in\mathcal{Q}(\mathbf{a})}\varepsilon_q.
\end{align}
\end{theorem}

\begin{proof}
A semantic false authorization requires at least one prerequisite in $\mathcal{Q}(\mathbf{a})$ to be false while passing Eq.~\ref{eq-gate-prerequisite}. By Eq.~\ref{eq-calibration-assumption} and the union bound, the probability of this event is at most $\sum_q(\delta_q+\varepsilon_q)$. Equation~\ref{eq-gate-risk-budget} gives $\sum_q\delta_q\le\bar\delta_B$. Invariant physical violations are independently rejected by the authoritative verifier and therefore do not increase the semantic bound under the stated verifier assumption.
\end{proof}

This result separates knowledge uncertainty from physical verification: belief quality determines the semantic authorization risk, while invariant safety remains protected by the independent verifier.

\subsection{Finite Belief Correction}
\begin{theorem}[Finite correction after poisoning]
\label{thm-finite-correction}
Suppose an attack on assertion $q$ ends at time $t_a$ and, from that time onward, the attacked and reference trajectories receive the same uncompromised evidence and verified feedback. Let
\begin{align}
\Delta_a=|L_{t_a}^A(q)-L_{t_a}^0(q)|.
\end{align}
Under $0<\lambda_f<1$, the belief difference after $k$ additional updates satisfies
\begin{align}
\label{eq-thm-correction-decay}
|b_{t_a+k}^A(q)-b_{t_a+k}^0(q)|
\le\frac{\lambda_f^k\Delta_a}{4}.
\end{align}
Consequently, for tolerance $\epsilon_{corr}>0$, a sufficient correction time is
\begin{align}
\label{eq-thm-correction-time}
T_{corr}(q)
\le
\left\lceil
\frac{\ln\!\left(\max\{1,\Delta_a/(4\epsilon_{corr})\}\right)}
{\ln(1/\lambda_f)}
\right\rceil.
\end{align}
\end{theorem}

\begin{proof}
After $t_a$, no additional adversarial term remains and both trajectories receive identical new evidence. Subtracting their instances of Eq.~\ref{eq-patb-update} gives $|\Delta_{t+1}|=\lambda_f|\Delta_t|$, hence $|\Delta_{t_a+k}|=\lambda_f^k\Delta_a$. Applying the $1/4$ Lipschitz constant of the sigmoid yields Eq.~\ref{eq-thm-correction-decay}. Solving $\lambda_f^k\Delta_a/4\le\epsilon_{corr}$ for $k$ gives Eq.~\ref{eq-thm-correction-time}.
\end{proof}

The bound is conservative because consistent post-attack evidence can correct the belief faster than temporal forgetting alone.

\subsection{Bounded Policy Drift}
\begin{theorem}[Trust-gated deployed-policy drift]
\label{thm-policy-drift}
Consider attacked and reference deployments that start from the same stable policy and undergo at most $K$ successful promotion events. At promotion $k$, let the attacked and reference candidates satisfy the KL constraints
\begin{align}
D_{\rm KL}(\Pi_k^A\|\Pi_{k-1}^A)&\le\epsilon_k^A,\\
D_{\rm KL}(\Pi_k^0\|\Pi_{k-1}^0)&\le\epsilon_k^0,
\end{align}
where $\epsilon_k^A=\epsilon_\pi(\bar\Gamma_k^A)$ is enforced by Algorithm~\ref{alg-trust-policy}. Then the total-variation distance after $K$ promotions is bounded by
\begin{align}
\label{eq-thm-tv-policy}
D_{TV}(\Pi_K^A,\Pi_K^0)
\le
\sum_{k=1}^{K}
\left(\sqrt{\frac{\epsilon_k^A}{2}}+
\sqrt{\frac{\epsilon_k^0}{2}}\right).
\end{align}
For a finite action head whose reference policy probabilities are lower bounded by $p_{min}>0$, this further implies
\begin{align}
\label{eq-thm-kl-policy}
D_{\rm KL}(\Pi_K^A\|\Pi_K^0)
\le
\frac{4}{p_{min}}
\left[
\sum_{k=1}^{K}
\left(\sqrt{\frac{\epsilon_k^A}{2}}+
\sqrt{\frac{\epsilon_k^0}{2}}\right)
\right]^2.
\end{align}
\end{theorem}

\begin{proof}
Pinsker's inequality gives
$D_{TV}(\Pi_k^A,\Pi_{k-1}^A)\le\sqrt{\epsilon_k^A/2}$ and the analogous bound for the reference policy. Applying the triangle inequality recursively from the common initial policy yields Eq.~\ref{eq-thm-tv-policy}. For a finite action distribution with $\Pi_K^0(a)\ge p_{min}$, $D_{\rm KL}(P\|Q)\le\chi^2(P\|Q)\le\|P-Q\|_1^2/p_{min}=4D_{TV}^2(P,Q)/p_{min}$, giving Eq.~\ref{eq-thm-kl-policy}.
\end{proof}

Because $\epsilon_\pi(\cdot)$ is nondecreasing, lower-trust update batches cannot receive a larger promotion radius. The theorem therefore formalizes the role of stable--candidate separation: adversarial evidence may alter candidate learning, but each deployed change remains explicitly bounded and must also pass shadow validation and the recovery gate.

\subsection{Scope of the Guarantees}
The four results are conditional security guarantees rather than a global convergence theorem for heterogeneous MARL. They require the trusted components specified in Section~\ref{subsec-threat-model}: authenticated source identity, correct implementation of the provenance cap and belief update, and an uncompromised authoritative verifier. They do not protect against arbitrary corruption of the TBox, verifier code, or all independent provenance groups simultaneously. These boundaries are evaluated explicitly in the experiments rather than hidden inside a general learning-convergence claim.
